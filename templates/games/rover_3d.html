# CodeQuest Rover Beacon Pathfinder from quest_grid import GridMatrix, RoverBot def navigate_to_beacon(rover: RoverBot, grid: GridMatrix) -> bool: """Autonomous loop for laser-safe beacon rendezvous.""" max_steps, elapsed = 30, 0 while rover.beacon.distance > 0 and elapsed < max_steps: # 1. Telemetry scan next tile step = grid.find_safe_step(rover.pos, rover.beacon.pos) if step.is_blocked: rover.recalibrate_radar(angle=45) continue # 2. Commit locomotion step rover.move_to(step.coords) elapsed += 1 # 3. Energy extraction check if grid.has_gem(rover.pos): rover.collect_gem() return rover.beacon.distance == 0
